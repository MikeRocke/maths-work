\result{Prove that if $x$ is an odd integer, then $9x+5$ is even.}

\begin{proof}
  Let $x=2k+1$ such that $k \in \mathbb{Z}$
  \begin{align*}
    9x+5 &= 9(2k+1) + 5\\
    &= 18k + 14\\
    &= 2(9k+7)
  \end{align*}, since $9k+7$ is an integer, thus $9x+5$ is even 
\end{proof}

\result{Prove that if x is an even integer, then $5x-3$ is an odd integer}

\begin{proof}
  Let $x=2k$, such that $k \in \mathbb{Z}$
  \begin{align*}
    5x - 3 &= 5(2k) - 3\\
           &= 10k - 3\\
           &= (10k - 2) - 1 \\
           &= 2(5k - 1) - 1 \\
  \end{align*}
  since $5k-1$ is an integer, then $5x-3$ is an odd integer
\end{proof}

\result{Let $n \in \mathbb{Z}$ Prove that $1-n^2 > 0$ then $3n-2$ is an even integer}
\begin{proof}
  Proving by contrapositive of the implication,  $1-n^2 > 0$ then $3n-2$ is an even integer, we have $3n-2$ is an odd number gives $1-n^2 \le 0$.
  let $n = 2k + 1$ then we have
  \begin{align*}
    1 - n^2 &= 1 - {(2k + 1)}^2 \\
            &= 1 - 4k^2 - 4k - 1 \\
            &= -4k^2 - 4k
  \end{align*}
 which $-4k^2 -4k$ is an integer, always less than or equal to 0 
\end{proof}

\result{Let $x \in \mathbb{Z}$  Prove that if $7x+5$ is odd, then $x$ is even}
\begin{proof}
  Proving by contrapositive of the implication, if $7x+5$ is odd, then $x$ is even, is if $x$ is odd, then $7x+5$ is even. Let $x = 2k + 1$ such that $k \in \mathbb{Z}$.
  \begin{align*}
    7x + 5 &= 7(2k + 1) + 5 \\
           &= 14k + 7 + 5 = 14k + 12 \\
           &= 2(7k + 6)
  \end{align*}
  which $2(7k + 6)$ is an even integer
\end{proof}
